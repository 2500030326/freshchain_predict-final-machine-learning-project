MATHEMATICAL FORMULA:
g(t) = [R(t) - R(T_t)] / (|T_t| - 1)

Measures the cost-to-complexity ratio of collapsing subtree T_t into a single leaf t. The node with minimal g(t) represents the weakest link and is pruned first.

💻 Python Implementation · Production MLOps: Drift Monitoring & FastAPI Serving (M6)
from scipy.stats import ks_2samp
import numpy as np
from fastapi import FastAPI
from pydantic import BaseModel
import onnxruntime as ort

# 1. Production Data Drift Detection via Kolmogorov-Smirnov Test
def detect_feature_drift(baseline_data, production_data, alpha=0.01):
    stat, p_val = ks_2samp(baseline_data, production_data)
    drift_detected = p_val < alpha
    return {"ks_statistic": round(float(stat), 4), "p_value": round(float(p_val), 4), "drift": drift_detected}

# 2. Ultra Low-Latency Model Serving with ONNX Runtime & FastAPI
app = FastAPI(title="Produce Freshness Prediction API")
session = ort.InferenceSession("models/champion_model.onnx")

class ProducePayload(BaseModel):
    Temp: float
    Humid: float
    Light: float
    CO2: float

@app.post("/predict")
def serve_prediction(item: ProducePayload):
    inputs = np.array([[item.Temp, item.Humid, item.Light, item.CO2]], dtype=np.float32)
    ort_inputs = {session.get_inputs()[0].name: inputs}
    proba = session.run(None, ort_inputs)[1][0][1] # P(Good)
    return {"label": "Good" if proba >= 0.5 else "Bad", "probability": round(float(proba), 4)}

💻 Python Implementation · Nested Stratified Cross-Validation (M5-2)
from sklearn.model_selection import StratifiedKFold, GridSearchCV, cross_val_score
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression

# Double-loop architecture eliminating hyperparameter selection bias
pipeline = Pipeline([
    ('scaler', StandardScaler()),
    ('model', LogisticRegression(max_iter=1000, random_state=42))
])

# Inner Loop: Tunes hyperparameter C on training folds only
inner_cv = StratifiedKFold(n_splits=3, shuffle=True, random_state=42)
grid_search = GridSearchCV(
    estimator=pipeline,
    param_grid={'model__C': [0.01, 0.1, 1.0, 10.0]},
    cv=inner_cv,
    scoring='f1'
)

# Outer Loop: Unbiased generalization estimation on held-out test folds
outer_cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
nested_f1_scores = cross_val_score(grid_search, X, y, cv=outer_cv, scoring='f1')

# Total model fits: K_out * (K_in * |Lambda| + 1) = 5 * (3 * 4 + 1) = 65 fits
print(f"Unbiased Nested F1: {nested_f1_scores.mean():.4f} (+/- {nested_f1_scores.std()*2:.4f})")

💻 Python Implementation · F-Beta Harmonic Mean & Huber Loss (M5-3)
import numpy as np

def compute_f_beta(precision: float, recall: float, beta: float = 1.0) -> float:
    """Generalized F-beta: (1 + beta^2) * (P * R) / (beta^2 * P + R)"""
    if precision + recall == 0:
        return 0.0
    b2 = beta ** 2
    return float((1 + b2) * (precision * recall) / (b2 * precision + recall))

def compute_huber_loss(residuals: np.ndarray, delta: float = 1.345) -> np.ndarray:
    """Huber Hybrid Loss: quadratic for |e| <= delta, linear for |e| > delta"""
    abs_e = np.abs(residuals)
    return np.where(abs_e <= delta, 0.5 * (residuals ** 2), delta * (abs_e - 0.5 * delta))

# Slide 9-11 Case Study: S08 outlier error e = $45k
residuals = np.array([3.0, 0.0, 5.0, 0.0, 5.0, 0.0, 5.0, 45.0, 2.0, 0.0])

💻 Python Implementation · ROC Threshold Trajectory & Pairwise AUC (M5-4)
from sklearn.metrics import confusion_matrix, roc_curve, roc_auc_score
import numpy as np

def evaluate_threshold(y_true, y_probs, tau=0.72, c_fp=1.0, c_fn=5.0):
    y_pred = (y_probs >= tau).astype(int)
    cm = confusion_matrix(y_true, y_pred, labels=[0, 1])
    tn, fp, fn, tp = cm[0,0], cm[0,1], cm[1,0], cm[1,1]
    tpr = tp / (tp + fn) if (tp + fn) > 0 else 0.0
    fpr = fp / (fp + tn) if (fp + tn) > 0 else 0.0
    youden_j = tpr - fpr
    dist_corner = np.sqrt((1.0 - tpr)**2 + fpr**2)
    expected_loss = c_fp * fp + c_fn * fn
    return {"TP": tp, "FP": fp, "TN": tn, "FN": fn, "TPR": tpr, "FPR": fpr, "Youden_J": youden_j, "Loss": expected_loss}

# Optimal Cutoff tau* = 0.72 achieves Youden's J = 1.0000 (Perfect separation)
# Pairwise Ranking AUC: U / (N_pos * N_neg) = 25 / (5 * 5) = 1.0000

1. The 100,000-Sample FPR Dilution Proof (Slide 3)

Why ROC-AUC is dangerously deceptive when positive prevalence π < 5%:

Setup: 100 Positives, 99,000 Negatives (π = 0.10%).
Model Hits: 90 True Positives, 990 False Positives.
• Recall: 90 / 100 = 90.0%
• FPR: 990 / 99,000 = 1.0% (ROC curve looks world-class!)
• Precision: 90 / (90 + 990) = 8.33% (11 false alarms for every 1 real hit!)

Conclusion: The massive pool of 99,000 true negatives dilutes the FPR denominator, masking high false alarm volume. PR-AUC directly reflects operational reality.

💻 Python Implementation · Platt Scaling Sigmoid & Isotonic PAVA (M5-6)
import numpy as np
from sklearn.isotonic import IsotonicRegression

# 1. Platt Scaling: p_cal = 1 / (1 + exp(-(a*f + b))), where f = ln(p/(1-p))
def platt_scale_manual(p_raw, a=1.773, b=0.789):
    eps = 1e-12
    p_clipped = np.clip(p_raw, eps, 1 - eps)
    f = np.log(p_clipped / (1 - p_clipped)) # Log-odds logit
    return 1.0 / (1.0 + np.exp(-(a * f + b)))

# S7 Worked Example: p = 0.60 -> f = ln(1.5) = 0.405 -> p_cal = 0.819

💻 Python Implementation · Search Budget Solver & GP Expected Improvement (M5-7)
import math
import numpy as np
from scipy.stats import norm

# Slide 10 Search Budget Theorem (Bergstra-Bengio):
# P(at least one in top q) = 1 - (1 - q)^n
# For q = 0.05 and n = 60: P = 1 - (0.95)^60 = 0.9539 (95.4%)
# Minimum budget for 95% confidence: n >= ln(0.05) / ln(0.95) = 58.4 -> n = 59

def compute_expected_improvement(mu, sigma, f_best):
    """Expected Improvement: EI(theta) = (mu - f_best)*Phi(Z) + sigma*phi(Z)"""
    z = (mu - f_best) / (sigma + 1e-9)
    ei = (mu - f_best) * norm.cdf(z) + sigma * norm.pdf(z)
    return np.maximum(ei, 0.0)

import numpy as np
from scipy.cluster.hierarchy import linkage, dendrogram, fcluster

# 5 Points: A=2, B=4, C=10, D=12, E=18
X = np.array([2.0, 4.0, 10.0, 12.0, 18.0]).reshape(-1, 1)

# Ward linkage variance increase: Delta_J = (|A|*|B| / (|A| + |B|)) * ||mu_A - mu_B||^2
# Height: h = sqrt(Delta_J)
# Merge Sequence:
# Step 1: (A,B) -> h = 1.41
# Step 2: (C,D) -> h = 1.41
# Step 3: ({CD}, E) -> h = 5.72
# Step 4: ({AB}, {CDE}) -> h = 11.32 (Root)
# Optimal Cut: Largest gap is 11.32 - 5.72 = 5.60 -> Cut at h = 8.0 yields K = 2 clusters!
Z = linkage(X, method='ward')
clusters = fcluster(Z, t=2, criterion='maxclust')
